    89→function getRiskColor(level: string): string {
    90→  switch (level) {
    91→    case 'LOW': return 'text-emerald-600';
    92→    case 'MEDIUM': return 'text-amber-600';
    93→    case 'HIGH': return 'text-orange-600';
    94→    case 'CRITICAL': return 'text-red-600';
    95→    default: return 'text-gray-600';
    96→  }
    97→}
    98→
    99→function getRiskBg(level: string): string {
   100→  switch (level) {
   101→    case 'LOW': return 'bg-emerald-50 border-emerald-200';
   102→    case 'MEDIUM': return 'bg-amber-50 border-amber-200';
   103→    case 'HIGH': return 'bg-orange-50 border-orange-200';
   104→    case 'CRITICAL': return 'bg-red-50 border-red-200';
   105→    default: return 'bg-gray-50 border-gray-200';
   106→  }
   107→}
   108→
   109→function getRiskBadgeVariant(level: string): 'default' | 'secondary' | 'destructive' | 'outline' {
   110→  switch (level) {
   111→    case 'LOW': return 'secondary';
   112→    case 'MEDIUM': return 'outline';
   113→    case 'HIGH': return 'default';
   114→    case 'CRITICAL': return 'destructive';
   115→    default: return 'outline';
   116→  }
   117→}
   118→
   119→function formatKES(n: number): string {
   120→  return `KSh ${(n / 1000).toFixed(0)}K`;
   121→}
   122→
   123→function timeAgo(dateStr: string): string {
   124→  const diff = Date.now() - new Date(dateStr).getTime();
   125→  const mins = Math.floor(diff / 60000);
   126→  if (mins < 60) return `${mins}m ago`;
   127→  const hrs = Math.floor(mins / 60);
   128→  if (hrs < 24) return `${hrs}h ago`;
   129→  return `${Math.floor(hrs / 24)}d ago`;
   130→}
   131→
   132→function getRecIcon(rec: string) {
   133→  switch (rec) {
   134→    case 'APPROVE_LOAN': return <CheckCircle2 className="h-4 w-4 text-emerald-600" />;
   135→    case 'REVIEW_MANUALLY': return <AlertTriangle className="h-4 w-4 text-amber-600" />;
   136→    case 'REJECT_LOAN': return <XCircle className="h-4 w-4 text-red-600" />;
   137→    default: return null;
   138→  }
   139→}
