    89→    89→function getRiskColor(level: string): string {
    90→    90→  switch (level) {
    91→    91→    case 'LOW': return 'text-emerald-600';
    92→    92→    case 'MEDIUM': return 'text-amber-600';
    93→    93→    case 'HIGH': return 'text-orange-600';
    94→    94→    case 'CRITICAL': return 'text-red-600';
    95→    95→    default: return 'text-gray-600';
    96→    96→  }
    97→    97→}

    99→    99→function getRiskBg(level: string): string {

   101→   101→    case 'LOW': return 'bg-emerald-50 border-emerald-200';
   102→   102→    case 'MEDIUM': return 'bg-amber-50 border-amber-200';
   103→   103→    case 'HIGH': return 'bg-orange-50 border-orange-200';
   104→   104→    case 'CRITICAL': return 'bg-red-50 border-red-200';

   109→   109→function getRiskBadgeVariant(level: string): 'default' | 'secondary' | 'destructive' | 'outline' {

   119→   119→function formatKES(n: number): string {

   122→   122→
   123→   123→function timeAgo(dateStr: string): string {
   124→   124→  const diff = Date.now() - new Date(dateStr).getTime();
   125→   125→  const mins = Math.floor(diff / 60000);
   126→   126→  if (mins < 60) return `${mins}m ago`;
   127→   127→  const hrs = Math.floor(mins / 60);
   128→   128→  if (hrs < 24) return `${hrs}h ago`;
   129→   129→  return `${Math.floor(hrs / 24)}d ago`;
   130→   130→}
